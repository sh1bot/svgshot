cmake_minimum_required(VERSION 3.20)
project(svgshot_capture_linux LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CAPTURE REQUIRED IMPORTED_TARGET atspi-2 x11 libpng json-c zlib)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  OUTPUT_VARIABLE SVGSHOT_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT SVGSHOT_COMMIT)
  set(SVGSHOT_COMMIT unknown)
endif()
add_executable(svgshot-capture-linux main.cpp)
target_compile_features(svgshot-capture-linux PRIVATE cxx_std_17)
target_compile_options(svgshot-capture-linux PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(svgshot-capture-linux PRIVATE SVGSHOT_COMMIT="${SVGSHOT_COMMIT}")
target_link_libraries(svgshot-capture-linux PRIVATE PkgConfig::CAPTURE)
