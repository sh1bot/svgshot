cmake_minimum_required(VERSION 3.20)
project(svgshot_capture LANGUAGES C CXX)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  OUTPUT_VARIABLE SVGSHOT_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT SVGSHOT_COMMIT)
  set(SVGSHOT_COMMIT unknown)
endif()
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build with Visual Studio and the Windows 10/11 SDK")
endif()
add_executable(svgshot-capture-win main.cpp)
add_library(capture-deflate STATIC vendor/miniz/miniz.c vendor/miniz/miniz_tdef.c)
set_property(TARGET svgshot-capture-win capture-deflate PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(svgshot-capture-win PRIVATE SVGSHOT_COMMIT="${SVGSHOT_COMMIT}")
target_include_directories(capture-deflate PUBLIC vendor/miniz)
target_compile_definitions(capture-deflate PUBLIC MINIZ_NO_INFLATE_APIS MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
target_link_libraries(svgshot-capture-win PRIVATE capture-deflate)
target_compile_features(svgshot-capture-win PRIVATE cxx_std_20)
target_compile_definitions(svgshot-capture-win PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(svgshot-capture-win PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(svgshot-capture-win PRIVATE uiautomationcore ole32 oleaut32 oleacc user32 gdi32 dwmapi d3d11 dxgi windowscodecs windowsapp shell32 shcore uuid advapi32 comctl32)
